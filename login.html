Login

async function login() { const email = document.getElementById("email").value.trim(); const password = document.getElementById("password").value.trim(); const status = document.getElementById("login-status"); status.innerText = "Logging in..."; try { const response = await fetch("https://godmode-backend2.onrender.com/auth/login", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ email, password }) }); const data = await response.json(); if (!response.ok) { status.innerText = data.detail || "Login failed"; return; } // Save token for premium modes localStorage.setItem("godmode_token", data.access_token); status.innerText = "Login successful!"; window.location.href = "dashboard.html"; // redirect after login } catch (err) { status.innerText = "Network error"; } } const token = localStorage.getItem("godmode_token"); if (!token) { window.location.href = "login.html"; }
